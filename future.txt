#include <iostream>
#include <string>
#include <vector>
#include <CoreFoundation/CoreFoundation.h>

std::string getLocalizedMoviesFolderName() {
    // 1. Find the path to the Movies directory using standard system search paths
    CFRange range = CFRangeMake(0, 0);
    CFURLRef homeDirURL = CFCopyHomeDirectoryURL();
    if (!homeDirURL) return "Movies";

    CFURLRef moviesURL = CFURLCreateCopyAppendingPathComponent(kCFAllocatorDefault, homeDirURL, CFSTR("Movies"), true);
    CFRelease(homeDirURL);
    if (!moviesURL) return "Movies";

    std::string result = "Movies";
    CFTypeRef localizedNameRef = nullptr;

    // 2. Query the localized name resource key
    if (CFURLCopyResourcePropertyForKey(moviesURL, kCFURLLocalizedNameKey, &localizedNameRef, nullptr)) {
        if (localizedNameRef && CFGetTypeID(localizedNameRef) == CFStringGetTypeID()) {
            CFStringRef cfStr = (CFStringRef)localizedNameRef;

            // Convert CFString to std::string (UTF-8)
            CFIndex length = CFStringGetLength(cfStr);
            CFIndex maxSize = CFStringGetMaximumSizeForEncoding(length, kCFStringEncodingUTF8) + 1;
            std::vector<char> buffer(maxSize);

            if (CFStringGetCString(cfStr, buffer.data(), maxSize, kCFStringEncodingUTF8)) {
                result = std::string(buffer.data());
            }
        }
        if (localizedNameRef) CFRelease(localizedNameRef);
    }

    CFRelease(moviesURL);
    return result;
}

int main() {
    std::cout << "Localized Movies folder name: " << getLocalizedMoviesFolderName() << std::endl;
    return 0;
}
